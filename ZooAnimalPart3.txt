package AlarconJavaZooPart3;

public class ZooAnimal {
    private String name;
    private String type;
    private int age;
    private boolean hungry;

    public ZooAnimal() {
    }

    public ZooAnimal(String name, String type, int age, boolean hungry) {
        this.name = name;
        this.type = type;
        this.age = age;
        this.hungry = hungry;
    }

    public void setName(String name) { this.name = name; }
    public String getName() { return this.name; }

    public void setType(String type) { this.type = type; }
    public String getType() { return this.type; }

    public void setAge(int age) { this.age = age; }
    public int getAge() { return this.age; }

    public boolean isHungry() { return this.hungry; }

    public void sleep() {
        System.out.println(name + " is sleeping.");
    }

    // Overloaded sleep method (Page 1)
    public void sleep(int hours) {
        System.out.println(name + " has slept for " + hours + " hours.");
    }

    public void eat() {
        System.out.println(name + " is eating.");
    }

    // Overloaded eat method (Page 1)
    public void eat(String food) {
        System.out.println(name + " is eating " + food + ".");
    }
}

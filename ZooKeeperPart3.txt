package AlarconJavaZooPart3;

public class ZooKeeper {
    private String name;
    private String title;
    private double payRate;
    private boolean degree;

    public ZooKeeper() {
    }

    public ZooKeeper(String name, String title, double payRate, boolean degree) {
    	super();
    	this.name = name;
        this.title = title;
        this.payRate = payRate;
        this.degree = degree;
    }

    public void setName(String name) { this.name = name; }
    public String getName() { return this.name; }

    public void setTitle(String title) { this.title = title; }
    public String getTitle() { return this.title; }

    public void setPayRate(double payRate) { this.payRate = payRate; }
    public double getPayRate() { return this.payRate; }

    public void setDegree(boolean degree) { this.degree = degree; }
    public boolean hasDegree() { return this.degree; }

    public void feedAnimal() {
        System.out.println(name + " is feeding a ZooAnimal.");
    }

    // Overloaded feedAnimal method (Page 1)
    public void feedAnimal(ZooAnimal animal, String food) {
        System.out.println(name + " is feeding " + animal.getName() + ".");
        animal.eat(food);
    }

    public void cleanCage() {
        System.out.println(name + " is cleaning the cage.");
    }

    // payMe method (Page 2)
    public double payMe(int hours) {
        return payRate * hours;
    }
}
